.htaccess

# BEGIN WordPress
<IfModule mod_rewrite.c>
RewriteEngine On
RewriteBase /

# redirect http to https (added 2019)
RewriteCond %{HTTPS} !=on
RewriteRule ^ https://%{HTTP_HOST}%{REQUEST_URI} [L,R=301]

# redirect www to non-www (hosting support told us to add this)
RewriteCond %{HTTP_HOST} ^www\.(.+)$ [NC]
RewriteRule ^ https://%1%{REQUEST_URI} [L,R=301]

# main wordpress rules
RewriteRule ^index\.php$ - [L]
RewriteCond %{REQUEST_FILENAME} !-f
RewriteCond %{REQUEST_FILENAME} !-d
RewriteRule . /index.php [L]
</IfModule>
# END WordPress


# block access to sensitive files
<FilesMatch "\.(env|log|ini|bak|sql|swp)$">
Order allow,deny
Deny from all
</FilesMatch>

# protect .htaccess itself
<Files .htaccess>
Deny from all
</Files>


# custom error pages
ErrorDocument 404 /errors/404.html
ErrorDocument 500 /errors/500.html


# PHP settings (shared hosting, don't ask...)
php_flag display_errors Off
php_value memory_limit 256M
php_value upload_max_filesize 64M
php_value post_max_size 64M


# cache static files
<IfModule mod_expires.c>
ExpiresActive On
ExpiresByType image/jpg "access plus 1 year"
ExpiresByType image/jpeg "access plus 1 year"
ExpiresByType image/png "access plus 1 year"
ExpiresByType image/gif "access plus 1 year"
ExpiresByType text/css "access plus 1 month"
ExpiresByType application/javascript "access plus 1 month"
</IfModule>


# gzip (may already be enabled by host)
<IfModule mod_deflate.c>
AddOutputFilterByType DEFLATE text/html text/plain text/css application/javascript
</IfModule>


# old redirects from previous site
Redirect 301 /old-page.html /new-page/
Redirect 301 /about-us.html /about/


# temporary fix - remove later
#RewriteRule ^api/ - [F]


# disable directory listing
Options -Indexes

# END FILE
